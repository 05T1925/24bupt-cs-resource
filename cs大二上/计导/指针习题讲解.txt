6-33 实验10_8_设计函数 void delcharfun(char *str,char ch)
设计函数 void delcharfun(char *str,char ch)实现从字符串str中删除指定的字符ch。同一字母的大、小写按不同字符处理。
思路：类似整数数组的删除元素设计，注意可能删除多次。
void delcharfun(char *str,char ch)
{	int i,j;
	for (i=0; str[i]!='\0'; )
		if (str[i]==ch){
			for(j=i+1; str[j]!='\0'; j++) 
				str[j-1]=str[j];
			str[j-1]='\0'; //删除元素后修正数组末尾结束符
		} else i++;
}

#include<stdlib.h>
int main()
{
	int n;

	for(i=0;i<n;i++)
	{
		scanf("%d",arr+i);   //&arr[i], arr+i
		arr[i] *= -10;
	}

void cal(int arr[], int size)
{
	int i;
	int sum=0;
	int maxn=-1e9,minn=1e9;
	for(i=0;i<size;i++)
	{
		scanf("%d",arr+i);
		sum+=arr[i];
		maxn=maxn>arr[i]?maxn:arr[i];
		minn=minn<arr[i]?minn:arr[i];
	}

	printf("%.2f\n",(double)sum/size);
	printf("%d\n",maxn);
	printf("%d\n",minn);
}

void fill(char ** strPtr , int n)
{
	int i,j;
	for(i=0;i<n;i++)
	{
		for(j=0;j<n+i;j++)
			*(*(strPtr+i)+j)=' ';
		*(*(strPtr+i)+n+i+1)='\0';

	{
		for(j=0;j<2*n-i;j++)

		*(*(strPtr+n+i)+(2*n-i))='/';

你要按照如下要求设计分割单词子函数int getString( char * source , char *strPtr[] ) ;

2.把字符串中单词结束后的空格改为“\0”,返回单词个数。
int getString(char * source, char *strPtr[] ){
    int i,num,flag;
    for (i=0,num=0,flag=0; source[i]!='\0'; i++){
        if (source[i]!=' '){
        	if (flag==0)
        	    strPtr[num]=source+i;
            	    flag=1;        	
		}
        else if (flag==1){
        	source[i]='\0';
            num++; flag=0;
        }
    }

	} 
    return num;
}

7-56 实验10_9_指针数组进阶
已知正整数n，n的范围是1—100。你要从键盘读入n个字符串，每个字符串的长度不确定，但是n个字符串的总长度不超过100000。你要利用字符指针数组将这n个字符串按照ASCII码顺序进行升序排序，然后再打印到屏幕上。字符串中可能包含ASCII码中的任意字符，每个字符串以换行符结束输入。
要求：不允许定义如char str[100][100000];这样的二维数组，因为会极大的浪费内存空间。你应定义char str[100000];这样的存储空间，将n个字符串连续的存储在一维字符空间内，然后将这n个字符串的起始位置保存在字符指针数组中，再进行排序操作。
#include<stdio.h>
#include<string.h>
void  strSort(char * a[], int n);
int main()
{
    char    str[100005];
    char    *strPtr[100]={0};
    int     i, n, loc;
    
	scanf("%d\n",&n);
	for(i=0,loc=0;i<n;i++)
	{
    	strPtr[i]=str+loc;
		gets(str+loc);
    	loc += 1+strlen(str+loc);
	}
    strSort(strPtr,n);
    for( i = 0 ; i < n ; i++ )
        puts(strPtr[i]);
    
    return 0;    
}
void  strSort(char * a[], int n){
   char * pt;
    int i, j, min;
      /*使用选择排序, 外循环每循环一次,确定a[i]的值*/
    for(i=0; i<n-1; i++){
        min=i; 			
        for(j=i+1; j<n; j++) 	
            if(strcmp(a[min],a[j])>0)
                min=j; 	
	  	if(min!=i){ 		
			pt=a[min];
			a[min]=a[i];
			a[i]=pt;
		}
    }
}
